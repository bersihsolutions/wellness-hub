Supabase PostgreSQL Schema Setup Script

Copy and run this SQL script in your Supabase SQL Editor to generate all required tables and permissions: 

-- CREATE PROFILES TABLE
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name TEXT NOT NULL,
  department TEXT NOT NULL,
  email TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- CREATE GRATITUDE POSTS TABLE
CREATE TABLE IF NOT EXISTS public.gratitude_posts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
  author_name TEXT NOT NULL,
  content TEXT NOT NULL,
  likes INT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- CREATE MOOD LOGS TABLE
CREATE TABLE IF NOT EXISTS public.mood_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
  mood TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- CREATE USER BINGO COMPLETIONS TABLE
CREATE TABLE IF NOT EXISTS public.user_bingo (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
  tile_id INT NOT NULL,
  completed BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(user_id, tile_id)
);

-- CREATE CUSTOM ASSIGNED TASKS TABLE
CREATE TABLE IF NOT EXISTS public.custom_tasks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  task_text TEXT NOT NULL,
  assigned_date DATE DEFAULT CURRENT_DATE,
  active BOOLEAN DEFAULT TRUE
);

-- CREATE BURNOUT ASSESSMENTS TABLE
CREATE TABLE IF NOT EXISTS public.burnout_assessments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
  score INT NOT NULL,
  risk_level TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ENABLE ROW LEVEL SECURITY & PUBLIC PERMISSIONS
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.gratitude_posts ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.mood_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.user_bingo ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.custom_tasks ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.burnout_assessments ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public Read Profiles" ON public.profiles FOR SELECT USING (true);
CREATE POLICY "Public Insert Profiles" ON public.profiles FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Read Gratitude" ON public.gratitude_posts FOR SELECT USING (true);
CREATE POLICY "Public Insert Gratitude" ON public.gratitude_posts FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Read Moods" ON public.mood_logs FOR SELECT USING (true);
CREATE POLICY "Public Insert Moods" ON public.mood_logs FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Read Bingo" ON public.user_bingo FOR SELECT USING (true);
CREATE POLICY "Public Insert Bingo" ON public.user_bingo FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Read Tasks" ON public.custom_tasks FOR SELECT USING (true);
CREATE POLICY "Public Insert Tasks" ON public.custom_tasks FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Read Burnout" ON public.burnout_assessments FOR SELECT USING (true);
CREATE POLICY "Public Insert Burnout" ON public.burnout_assessments FOR INSERT WITH CHECK (true);
