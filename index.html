-- Execute SQL Script for Bersih Care Portal Supabase Dashboard

-- Profiles / Office Users
CREATE TABLE IF NOT EXISTS public.profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email TEXT UNIQUE NOT NULL,
    full_name TEXT NOT NULL,
    department TEXT NOT NULL,
    password_hash TEXT,
    bingo_points INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Daily Gratitude Note Wall
CREATE TABLE IF NOT EXISTS public.gratitude_posts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sender_name TEXT NOT NULL,
    recipient_name TEXT,
    message TEXT NOT NULL,
    hearts_count INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Daily Energy Pulse Logs
CREATE TABLE IF NOT EXISTS public.mood_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_email TEXT NOT NULL,
    mood TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- October Assigned Tasks & Broadcasts
CREATE TABLE IF NOT EXISTS public.assigned_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title TEXT NOT NULL,
    target_date DATE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Burnout & Work-Life Assessments
CREATE TABLE IF NOT EXISTS public.burnout_assessments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    score INT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
