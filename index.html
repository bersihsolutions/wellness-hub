To ensure cloud sync for gratitude posts and mood check-ins across all devices, execute this SQL script in your Supabase Dashboard SQL Editor:

-- 1. Create Gratitude Wall Table
CREATE TABLE IF NOT EXISTS gratitude_board (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    sender_name TEXT,
    recipient_name TEXT,
    message TEXT NOT NULL,
    category TEXT DEFAULT 'Appreciation',
    is_anonymous BOOLEAN DEFAULT false,
    likes INTEGER DEFAULT 0,
    user_email TEXT
);

-- 2. Create Mood Logs Table
CREATE TABLE IF NOT EXISTS mood_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    user_email TEXT,
    energy_level TEXT NOT NULL,
    notes TEXT
);

-- 3. Enable RLS and public access policies
ALTER TABLE gratitude_board ENABLE ROW LEVEL SECURITY;
ALTER TABLE mood_logs ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public Read Gratitude" ON gratitude_board FOR SELECT USING (true);
CREATE POLICY "Public Insert Gratitude" ON gratitude_board FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Read Mood" ON mood_logs FOR SELECT USING (true);
CREATE POLICY "Public Insert Mood" ON mood_logs FOR INSERT WITH CHECK (true);

Note: If Supabase tables are unavailable or network drops, Sakinah automatically preserves your state locally without interruption.
