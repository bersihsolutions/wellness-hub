Supabase DDL SQL Setup Script

Copy & paste into your Supabase SQL Editor to initialize all 8 required tables and seed data.

-- ===================================================
-- BERSIH CARE PORTAL - SUPABASE DDL & SEED SCRIPT
-- ===================================================

-- 1. Profiles Table
CREATE TABLE IF NOT EXISTS public.profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    full_name TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    department TEXT NOT NULL,
    password_hash TEXT NOT NULL,
    total_points INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. Gratitude Posts Table
CREATE TABLE IF NOT EXISTS public.gratitude_posts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
    user_name TEXT NOT NULL,
    content TEXT NOT NULL,
    likes_count INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Gratitude Likes Tracking Table
CREATE TABLE IF NOT EXISTS public.gratitude_likes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    post_id UUID REFERENCES public.gratitude_posts(id) ON DELETE CASCADE,
    user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
    UNIQUE(post_id, user_id)
);

-- 4. Mood Logs Table
CREATE TABLE IF NOT EXISTS public.mood_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
    department TEXT NOT NULL,
    energy_level TEXT NOT NULL,
    logged_date DATE DEFAULT CURRENT_DATE
);

-- 5. Bingo Tasks Master Table
CREATE TABLE IF NOT EXISTS public.bingo_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title TEXT NOT NULL,
    category TEXT NOT NULL,
    week_num INT NOT NULL,
    is_unlocked BOOLEAN DEFAULT true,
    assigned_date DATE,
    points INT DEFAULT 10
);

-- 6. User Bingo Progress Table
CREATE TABLE IF NOT EXISTS public.user_bingo_progress (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
    task_id UUID REFERENCES public.bingo_tasks(id) ON DELETE CASCADE,
    completed_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(user_id, task_id)
);

-- 7. Burnout Surveys Table
CREATE TABLE IF NOT EXISTS public.burnout_surveys (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
    department TEXT NOT NULL,
    score INT NOT NULL,
    status TEXT NOT NULL,
    logged_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 8. Daily Quotes Table
CREATE TABLE IF NOT EXISTS public.daily_quotes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    quote_text TEXT NOT NULL,
    source_type TEXT NOT NULL, -- 'general' | 'islamic'
    day_number INT UNIQUE NOT NULL
);
