Supabase SQL Setup Code

Copy and run this script in your Supabase SQL Editor to create the 4 required database tables and enable real-time replication: 

-- 1. Create Gratitude Notes Table
CREATE TABLE IF NOT EXISTS gratitude_notes (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  author TEXT,
  content TEXT NOT NULL,
  card_color TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Create Daily Pulse Table
CREATE TABLE IF NOT EXISTS daily_pulse (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  mood TEXT NOT NULL,
  emoji TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Create Quiet Zone Bookings Table
CREATE TABLE IF NOT EXISTS quiet_zone_bookings (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  pod_id TEXT NOT NULL,
  user_name TEXT NOT NULL,
  start_time TEXT NOT NULL,
  duration TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Enable Realtime Publications
ALTER PUBLICATION supabase_realtime ADD TABLE gratitude_notes;
ALTER PUBLICATION supabase_realtime ADD TABLE daily_pulse;
ALTER PUBLICATION supabase_realtime ADD TABLE quiet_zone_bookings;

-- 5. Open Anon Read/Write Permissions
ALTER TABLE gratitude_notes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read/write" ON gratitude_notes FOR ALL USING (true) WITH CHECK (true);

ALTER TABLE daily_pulse ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read/write" ON daily_pulse FOR ALL USING (true) WITH CHECK (true);

ALTER TABLE quiet_zone_bookings ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read/write" ON quiet_zone_bookings FOR ALL USING (true) WITH CHECK (true);
